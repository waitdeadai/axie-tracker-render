# Axie tracker backend — env template (SOTA-2026 Ronin on-chain access).
# Copy to backend/.env (local) or fold into ops/vps/runtime.cfg (VPS) and fill secrets.
# .env / .env.* are gitignored — never commit real secrets. This .example.txt IS committed.
# (Named *.example.txt, not .env.example, to match the repo convention in
#  ops/vps/env.vps.example.txt and to stay clear of secret-path write guards.)
#
# Access model: Discord login is GONE. Authorization is on-chain:
#   isAuthorized := hasActiveAccess(wallet) || isWhitelisted(wallet)
# The app only ever RECEIVES USDC to PAYMENT_WALLET_ADDRESS. It NEVER holds or
# requires the owner's private key — there is intentionally no *_PRIVATE_KEY here.

# ---------------------------------------------------------------------------
# Core runtime
# ---------------------------------------------------------------------------
PORT=4000
NODE_ENV=production
DATA_DIR=/data            # access.db lives here; back it with a persistent volume

# ---------------------------------------------------------------------------
# Auth secrets (REQUIRED in production — the app refuses to issue/verify JWTs
# or sign sessions without one of these). JWT carries the SIWE-verified wallet;
# authorization is recomputed live from on-chain access on every request.
# Generate each with:  openssl rand -hex 32
# ---------------------------------------------------------------------------
JWT_SECRET=REPLACE_ME_openssl_rand_hex_32
SESSION_SECRET=REPLACE_ME_openssl_rand_hex_32
JWT_EXPIRES_IN=7d
# SIWE_DOMAIN must equal the host the wallet signs against. If unset, the server
# derives it from FRONTEND_URL, then the request Host header. Set it explicitly
# in prod so a forwarded/spoofed Host can't widen the accepted domain.
SIWE_DOMAIN=axie.waitdead.com

# ---------------------------------------------------------------------------
# Ronin chain access (chainId 2020 / RON gas / ~2s blocks, ~6s fast finality)
# ---------------------------------------------------------------------------
# Public RPC works with zero third-party deps (rate-limited). Alternatives:
#   https://ronin.drpc.org  (public)
#   https://api-gateway.skymavis.com/rpc  (needs RONIN_RPC_API_KEY, sent as X-API-KEY)
RONIN_RPC_URL=https://api.roninchain.com/rpc
# Optional Sky Mavis API gateway key. Only sent (as X-API-KEY) when non-empty.
RONIN_RPC_API_KEY=
RONIN_CHAIN_ID=2020

# ---------------------------------------------------------------------------
# Payment target + token (all public, safe to commit as the example values)
# ---------------------------------------------------------------------------
# Receive-only owner wallet, EIP-55 checksummed 0x form. App NEVER spends from it.
PAYMENT_WALLET_ADDRESS=0x51a8318FBFf6DDFee50Ee0fb0f33AF02F34fF649
# USDC on Ronin is BRIDGED via Chainlink CCIP (NOT native Circle). 6 decimals.
# (The pre-2026 default 0x9433...471c3 was WRONG — eth_getCode returned 0x, not a
#  contract. Do not reintroduce it.)
USDC_CONTRACT=0x0b7007c13325c48911f73a2dad5fa5dcbf808adc

# ---------------------------------------------------------------------------
# Plan prices (USDC). amount charged = price * 10^USDC_DECIMALS.
# v1 is USDC-only. AXS pricing is a documented future extension (needs a live
# price oracle) and is intentionally NOT implemented — do not add AXS_* here.
# ---------------------------------------------------------------------------
PLANS_USDC_2WEEKS=2       # 14 days
PLANS_USDC_1MONTH=5       # 30 days
PLANS_USDC_3MONTH=12      # 90 days
PLANS_USDC_1YEAR=40       # 365 days

# ---------------------------------------------------------------------------
# Payment detection
#   PRIMARY (zero-dep, default): eth_getLogs poller over USDC Transfer events to
#     PAYMENT_WALLET_ADDRESS, counting only transfers at/below the finalized head.
#   OPTIONAL: Moralis Streams webhook (instant). Only verified when the secret is
#     set; needs a Moralis account + this backend reachable at a public URL:
#     POST https://axie.waitdead.com/api/payment/webhook/moralis
# ---------------------------------------------------------------------------
PAYMENT_POLL_ENABLED=true
PAYMENT_POLL_INTERVAL_MS=12000
# How far back to rescan on (re)start, in blocks. ~2s/block -> 1800 ~= 1h.
PAYMENT_POLL_LOOKBACK_BLOCKS=1800
# Moralis (optional). MORALIS_API_KEY is used when creating the Stream out-of-band;
# MORALIS_STREAM_SECRET is the signing secret used to verify the webhook header.
# Leave both empty to run poller-only.
MORALIS_API_KEY=
MORALIS_STREAM_SECRET=

# ---------------------------------------------------------------------------
# Admin (season whitelist seeding). Guards POST /api/admin/whitelist via the
